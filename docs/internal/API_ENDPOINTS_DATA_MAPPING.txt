REQUEST DATA:
{
  "developer": "dev1",
  "file_path": "src/auth/login.py",
  "function_name": "validate_user"
}

RESPONSE (Lock Acquired):
{
  "success": true,
  "message": "dev1 started editing",
  "lock_acquired": true,
  "state": "editing",
  "developer": "dev1"
}

RESPONSE (Lock Blocked - Queued):
{
  "success": false,

STATE MACHINE IMPACT:
├─ First dev: state = EDITING, current_editor = dev1
└─ 2nd+ dev: state = CONFLICT_WAITING, queued_at = NOW(), queue_position auto-calc

STORED DATA:
├─ State machine state_history entry
├─ Development memory INTENT_DECLARED event
└─ Development memory RESOURCE_CLAIMED event

📍 ENDPOINT 3: /api/log_change (POST)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
When: Developer logs code changes
REQUEST DATA:
{
  "developer": "dev1",
  "file_path": "src/auth/login.py",
  "function_name": "validate_user",
  "old_code": "def validate_user(...):\n  ...",
  "new_code": "def validate_user(...):\n  ...",
  "feature_branch": "feature/auth-refactor",
  "verbal_description": "Refactored validation logic"
}

RESPONSE:
{
  "id": "2026-09-18T10:45:00.123456Z",
  "developer": "dev1",
  "file": "src/auth/login.py",
  "function": "validate_user",
  "branch": "feature/auth-refactor",
  "timestamp": "2026-09-18T10:45:00Z",
  "description": "Refactored validation logic",
  "old_lines": 12,
  "new_lines": 10,
  "line_changes": 4,
  "conflict_severity": "low",
  "related_changes": []
}

CONFLICT DETECTION LOGIC:
├─ Load previous changes from activity_log_storage/changes/
├─ Calculate diff between old_code and new_code
├─ Compare with previous developer's changes
├─ Calculate overlap_ratio = min_lines_changed / max_lines_changed
├─ Map overlap_ratio to severity:
│  ├─ 0.0 = "low"
│  ├─ < 0.5 = "medium"
│  └─ >= 0.5 = "high"
└─ If "high" → CREATE ESCALATION

STORED DATA:
├─ activity_log_storage/changes/{file}_{function}_{timestamp}.json
├─ If HIGH: activity_log_storage/escalations/{file}_{function}_escalation.json
└─ Development memory EVENT_CHANGE_LOGGED

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

└─ Update last_update timestamp

STORED DATA:
├─ activity_log_storage/approvals/{file}_{function}_approvals.json
└─ Must track BOTH approved_by and rejected_by lists

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
When: Check if merge is allowed
REQUEST PARAMS:
{
  "file_path": "src/auth/login.py",
  "function_name": "validate_user"
}

RESPONSE (Can Merge):
{
  "can_merge": true,
  "status": {
    "reason": "All developers approved",
    "approved_by": ["dev1", "dev3"],
    "status": "APPROVED"
  }
}

RESPONSE (Cannot Merge - Pending):
{
  "can_merge": false,
  "status": {
    "reason": "Waiting for approvals",
    "approved_by": ["dev1"],
    "waiting_for": ["dev2", "dev3"],
    "status": "PENDING_APPROVAL"
  }
}

RESPONSE (Cannot Merge - Rejected):
{
  "can_merge": false,
  "status": {
    "reason": "Merge REJECTED by developers",
    "rejected_by": ["dev2"],
    "approved_by": ["dev1"],
    "status": "REJECTED"
  }
}

MERGE CHECK LOGIC:
├─ Find all changes for file::function
├─ Check for HIGH conflicts (conflict_severity="high")
├─ If no HIGH conflicts → can_merge = true
├─ If HIGH conflicts exist:
│  ├─ Load approvals/{file}_{function}_approvals.json
│  ├─ If rejected_by != empty → can_merge = false (REJECTED)
│  ├─ If approved_by == all_developers → can_merge = true
│  └─ Else → can_merge = false (PENDING_APPROVAL)
└─ Return detailed status

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

================================================================================
CRITICAL VERIFICATION POINTS FOR DATA MODELS
================================================================================

✅ Queue Timestamp Ordering (MOST CRITICAL):
   File: workflow_state_machine.py (waiting_developers)
   Test: 
   - Dev2 joins at T1 (10:35:01Z)
   - Dev3 joins at T2 (10:36:00Z)
   - Dev4 joins at T3 (10:37:15Z)
   - When Dev1 finishes: MIN(T1, T2, T3) = Dev2 gets lock ← VERIFY THIS

✅ State Machine State Transitions:
   AVAILABLE → EDITING (First dev)
   EDITING → CONFLICT_WAITING (2nd+ dev joins)
   EDITING/CONFLICT_WAITING → BOTH_DONE (dev finishes)
   → No state change for subsequent queue members until they get lock

✅ Conflict Severity Calculation:
   overlap_ratio = min(prev_changes, curr_changes) / max(prev_changes, curr_changes)
   - 0.0 → "low"
   - 0.0 < x < 0.5 → "medium"
   - >= 0.5 → "high" → CREATE ESCALATION

✅ Approval Tracking:
   Both approved_by and rejected_by lists must be maintained
   If dev changes from rejected to approved: remove from rejected, add to approved
   Merge blocked if ANY developer rejected (unless they change approval)

✅ Lock Exclusivity:
   Only one current_editor at any time
   All others must be in waiting_developers with queued_at timestamp
   No "hidden" developers outside of these two states
